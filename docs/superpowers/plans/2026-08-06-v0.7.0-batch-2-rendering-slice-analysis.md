# v0.7.0 Batch 2 Rendering and Slice Analysis Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give every ready NetCDF RenderAsset consistent runtime rendering controls plus authoritative X/Y/Z single-plane analysis, statistics, and ZIP export.

**Architecture:** Keep SuperMap3D isolated in the same-origin iframe and upgrade its protocol to revisioned full-state messages. Resolve both candidate and legacy assets through one backend slice service; Vue consumes that service for ECharts and export without deriving values from pixels.

**Tech Stack:** Python 3.12, FastAPI, SQLAlchemy, NumPy, pandas, Vue 3, TypeScript, Element Plus, ECharts 5, Vitest, Playwright, SuperMap3D 12.1 `VoxelGridLayer3D`.

PaletteId = Literal["native-spectrum", "viridis", "turbo", "coolwarm", "grayscale"]

@dataclass(frozen=True)
class RenderProfile:
    property_name: str
    unit: str
    default_scale: ScaleMode
    default_palette: PaletteId
    log_available: bool
    value_range: tuple[float, float]
    filter_range: tuple[float, float]
    lighting: bool = True
    gradient_opacity: bool = True
    bounding_box: bool = True
    opacity: float = 1.0

def build_render_profile(source_kind, valid_min, valid_max, *, property_name, unit):
    log_available = valid_min > 0
    requested = "log" if source_kind == "builtin_legacy" else "linear"
    return RenderProfile(
        property_name=property_name,
        unit=unit or "unknown",
        default_scale=requested if requested != "log" or log_available else "linear",
        default_palette="native-spectrum" if source_kind == "builtin_legacy" else "viridis",
        log_available=log_available,
        value_range=(valid_min, valid_max),
        filter_range=(valid_min, valid_max),
    )

def _statistics(values: np.ndarray, mask: np.ndarray) -> dict[str, int | float | None]:
    valid = values[~mask]
    if valid.size == 0:
        return {"total_count": int(values.size), "valid_count": 0,
                "nodata_count": int(values.size), "min": None, "max": None,
                "mean": None, "std_population": None, "p10": None,
                "p50": None, "p90": None}
    q = np.quantile(valid, [0.1, 0.5, 0.9], method="linear")
    return {"total_count": int(values.size), "valid_count": int(valid.size),
            "nodata_count": int(values.size - valid.size), "min": float(valid.min()),
            "max": float(valid.max()), "mean": float(valid.mean()),
            "std_population": float(valid.std(ddof=0)), "p10": float(q[0]),
            "p50": float(q[1]), "p90": float(q[2])}
